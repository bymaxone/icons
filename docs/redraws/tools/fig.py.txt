import math,sys,json
H0='<svg xmlns="http://www.w3.org/2000/svg" width="%d" height="%d" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="%s" stroke-linecap="round" stroke-linejoin="round">'
def f(v): return ('%.2f'%v).rstrip('0').rstrip('.')
def P(p): return "%s %s"%(f(p[0]),f(p[1]))
def figure(hip, shoulder, arm_f, arm_b, leg_f, leg_b, r=1.75, gap=3.0):
    # arm_f etc: list of points after the joint (elbow, hand) / (knee, foot)
    dx,dy=shoulder[0]-hip[0],shoulder[1]-hip[1];L=math.hypot(dx,dy)
    head=(shoulder[0]+dx/L*(r+gap),shoulder[1]+dy/L*(r+gap))
    d=[]
    # back arm + spine + back leg: one polyline
    d.append("M"+"L".join(P(p) for p in list(reversed(arm_b))+[shoulder,hip]+leg_b))
    d.append("M"+"L".join(P(p) for p in [shoulder]+arm_f))
    d.append("M"+"L".join(P(p) for p in [hip]+leg_f))
    return head,"".join(d)
def svg(head,d,size=24,sw="1.5",r=1.75):
    return H0%(size,size,sw)+'\n  <circle cx="%s" cy="%s" r="%s"/>\n  <path d="%s"/>\n</svg>\n'%(f(head[0]),f(head[1]),f(r),d)
def build(name,**k):
    h,d=figure(**k);open(name+".svg","w").write(svg(h,d));return h,d
