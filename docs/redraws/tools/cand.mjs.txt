// Run from a copy named *.mjs placed inside the repository, so '@resvg/resvg-js' and 'svgo' resolve.
// node cand.mjs out192.png out1x.png file::label ...   (labels under every drawing; 192 px panel and true 24 px 1x panel, light over dark)
import { readFileSync, writeFileSync } from 'node:fs'
import { Resvg } from '@resvg/resvg-js'
const [out192, out1x, ...items] = process.argv.slice(2)
const parse = (f, fg) => {
  const t = readFileSync(f, 'utf8').trim()
  const inner = t.replace(/^<svg[^>]*>/, '').replace(/<\/svg>$/, '')
  const attrs = t.match(/^<svg([^>]*)>/)[1]
  const sw = attrs.match(/stroke-width="([^"]+)"/)?.[1] ?? '1.5'
  const filled = /fill="currentColor"/.test(attrs) && !/stroke=/.test(attrs)
  const g = filled ? `fill="${fg}"` : `fill="none" stroke="${fg}" stroke-width="${sw}" stroke-linecap="round" stroke-linejoin="round"`
  return { inner, g }
}
const render = (out, size, cell, labelSize, rows) => {
  const w = items.length * cell, h = rows.length * (size + labelSize * 2.2 + 8)
  let body = ''
  rows.forEach(([bg, fg], r) => {
    const y0 = r * (size + labelSize * 2.2 + 8)
    body += `<rect x="0" y="${y0}" width="${w}" height="${size + labelSize * 2.2 + 8}" fill="${bg}"/>`
    items.forEach((it, i) => {
      const [f, ...lab] = it.split('::'); const label = lab.join('::')
      const { inner, g } = parse(f, fg)
      const x = i * cell + (cell - size) / 2
      body += `<g transform="translate(${x} ${y0 + 4}) scale(${size / 24})" ${g}>${inner}</g>`
      body += `<text x="${i * cell + cell / 2}" y="${y0 + size + labelSize * 1.5 + 4}" font-size="${labelSize}" font-weight="700" text-anchor="middle" font-family="Helvetica, Arial, sans-serif" fill="${fg === '#111' ? '#c00' : '#ff8a80'}">${label}</text>`
    })
  })
  writeFileSync(out, new Resvg(`<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h}" viewBox="0 0 ${w} ${h}">${body}</svg>`, { font: { loadSystemFonts: true } }).render().asPng())
}
render(out192, 192, 230, 22, [['#fff', '#111']])
render(out1x, 24, 64, 11, [['#fff', '#111'], ['#16161a', '#f4f4f5']])
