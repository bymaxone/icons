import { readFileSync } from 'node:fs'
import { Resvg } from '/Users/maximiliano/Documents/MyApps/bymax-one/icons/node_modules/@resvg/resvg-js/index.js'
for (const f of process.argv.slice(2)) {
  const s = readFileSync(f, 'utf8').replace(/currentColor/g, '#000')
  const w = /width="16"/.test(s) ? 16 : 24
  const p = new Resvg(s, { fitTo: { mode: 'width', value: w } }).render()
  let a = 0; for (let i = 3; i < p.pixels.length; i += 4) a += p.pixels[i]
  console.log(f.split('/').pop(), ((a / 255 / (p.width * p.height)) * 100).toFixed(2) + '%')
}
