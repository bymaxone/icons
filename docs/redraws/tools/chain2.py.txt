import math
def f(v): return ('%.3f'%v).rstrip('0').rstrip('.')
class Ell:
    def __init__(s,cx,cy,a,b,phi): s.cx,s.cy,s.a,s.b,s.phi=cx,cy,a,b,math.radians(phi)
    def pt(s,t):
        x,y=s.a*math.cos(t),s.b*math.sin(t);c,sn=math.cos(s.phi),math.sin(s.phi)
        return (s.cx+x*c-y*sn,s.cy+x*sn+y*c)
def seg_dist(p,a,b):
    ax,ay=a;bx,by=b;dx,dy=bx-ax,by-ay;L2=dx*dx+dy*dy
    u=max(0,min(1,((p[0]-ax)*dx+(p[1]-ay)*dy)/L2)) if L2 else 0
    return math.hypot(p[0]-(ax+u*dx),p[1]-(ay+u*dy))
def poly_dist(p,poly):
    return min(seg_dist(p,poly[i],poly[i+1]) for i in range(len(poly)-1))
def ell_dist(p,E,N=720):
    return min(math.hypot(p[0]-q[0],p[1]-q[1]) for q in (E.pt(2*math.pi*i/N) for i in range(N)))
def crossings(A,B,N=3000):
    def inside(p):
        x,y=p[0]-B.cx,p[1]-B.cy;c,s=math.cos(-B.phi),math.sin(-B.phi)
        u,v=x*c-y*s,x*s+y*c;return (u/B.a)**2+(v/B.b)**2-1
    out=[];last=inside(A.pt(0))
    for i in range(1,N+1):
        t=2*math.pi*i/N;cur=inside(A.pt(t))
        if (cur>0)!=(last>0): out.append(A.pt(t))
        last=cur
    return out
def arcs(E,keep,N):
    # keep: list of bool per param sample; return path arcs for contiguous kept runs
    runs=[];i=0
    if all(keep): 
        p0,p1=E.pt(0),E.pt(math.pi)
        return "M%s %sA%s %s %s 1 1 %s %sA%s %s %s 1 1 %s %s"%(f(p0[0]),f(p0[1]),f(E.a),f(E.b),f(math.degrees(E.phi)),f(p1[0]),f(p1[1]),f(E.a),f(E.b),f(math.degrees(E.phi)),f(p0[0]),f(p0[1]))
    # rotate so that index 0 is a removed sample
    s0=keep.index(False)
    order=[(s0+k)%N for k in range(N)]
    cur=[]
    for idx in order:
        if keep[idx]: cur.append(idx)
        else:
            if cur: runs.append(cur);cur=[]
    if cur: runs.append(cur)
    d=""
    for r in runs:
        t0=2*math.pi*r[0]/N;span=(len(r)-1)*2*math.pi/N;t1=t0+span
        p0,p1=E.pt(t0),E.pt(t1);large=1 if span>math.pi else 0
        d+="M%s %sA%s %s %s %d 1 %s %s"%(f(p0[0]),f(p0[1]),f(E.a),f(E.b),f(math.degrees(E.phi)),large,f(p1[0]),f(p1[1]))
    return d
def link(cx=12.25,cy=12.25,a=6.4,b=3.6,sep=5.4,ang=-45,gap=3.0,obstacles=(),N=1440,scale=1.0,dx=0,dy=0,crossing=True):
    ca,sa=math.cos(math.radians(ang)),math.sin(math.radians(ang))
    A=Ell(cx+dx-sep*scale/2*ca,cy+dy-sep*scale/2*sa,a*scale,b*scale,ang)
    B=Ell(cx+dx+sep*scale/2*ca,cy+dy+sep*scale/2*sa,a*scale,b*scale,ang)
    X=crossings(A,B)
    d=""
    for E,O,Xp in ((A,B,X[0]),(B,A,X[1])):
        close=[ell_dist(E.pt(2*math.pi*i/N),O,360)<gap and math.hypot(E.pt(2*math.pi*i/N)[0]-Xp[0],E.pt(2*math.pi*i/N)[1]-Xp[1])<gap*2.2 for i in range(N)]
        near=min(range(N),key=lambda i:math.hypot(E.pt(2*math.pi*i/N)[0]-Xp[0],E.pt(2*math.pi*i/N)[1]-Xp[1]))
        rm=[False]*N
        if crossing and close[near]:
            i=near
            while close[i%N] and not rm[i%N]:
                rm[i%N]=True;i+=1
            i=near-1
            while close[i%N] and not rm[i%N]:
                rm[i%N]=True;i-=1
        for i in range(N):
            p=E.pt(2*math.pi*i/N)
            for ob in obstacles:
                if poly_dist(p,ob)<gap: rm[i]=True
        keep=[not x for x in rm]
        d+=arcs(E,keep,N)
    return d
